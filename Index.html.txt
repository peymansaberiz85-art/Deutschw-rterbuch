    <script>
        // LocalStorage database management
        let words = JSON.parse(localStorage.getItem('worterbuch_words') || '[]');
        let currentCardIndex = null;
        let currentCardWord = '';

        // Initialize App
        function initApp() {
            updateStats();
            loadNextCard();
        }

        // Theme Toggle
        function toggleTheme() {
            document.documentElement.classList.toggle('dark');
            localStorage.setItem('theme', document.documentElement.classList.contains('dark') ? 'dark' : 'light');
        }
        if (localStorage.getItem('theme') === 'dark' || (!localStorage.getItem('theme') && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
            document.documentElement.classList.add('dark');
        }

        // Modal Controls
        function openAboutModal() { document.getElementById('about-modal').classList.remove('hidden'); }
        function closeAboutModal() { document.getElementById('about-modal').classList.add('hidden'); }

        // AI Mock / Simulated Analyzer (Gemini 3 Flash integration stub)
        function aiAnalyzeWord() {
            const input = document.getElementById('word-input').value.trim();
            if(!input) {
                alert('لطفاً ابتدا یک واژه آلمانی وارد کنید.');
                return;
            }
            document.getElementById('ai-loading').classList.remove('hidden');
            
            setTimeout(() => {
                document.getElementById('ai-loading').classList.add('hidden');
                // Populate preview fields with smart estimations / standard entries
                document.getElementById('preview-german').value = input;
                document.getElementById('preview-persian').value = 'معنی واژه ' + input + ' (تحلیل شده با هوش مصنوعی)';
                document.getElementById('preview-example').value = 'Das ist ein Beispiel für ' + input + '.';
            }, 800);
        }

        // Save Word to Leitner Box 1
        function saveWordToLeitner() {
            const german = document.getElementById('preview-german').value.trim();
            const persian = document.getElementById('preview-persian').value.trim();
            const example = document.getElementById('preview-example').value.trim();

            if(!german || !persian) {
                alert('لطفاً فیلدهای آلمانی و فارسی را تکمیل کنید.');
                return;
            }

            const newWord = {
                id: Date.now(),
                german,
                persian,
                example,
                box: 1,
                lastReviewed: new Date().toISOString()
            };

            words.push(newWord);
            localStorage.setItem('worterbuch_words', JSON.stringify(words));
            
            // Reset inputs
            document.getElementById('word-input').value = '';
            document.getElementById('preview-german').value = '';
            document.getElementById('preview-persian').value = '';
            document.getElementById('preview-example').value = '';

            updateStats();
            loadNextCard();
            alert('واژه با موفقیت به خانه ۱ لایتنر افزوده شد.');
        }

        // Update Dashboard Stats
        function updateStats() {
            document.getElementById('stat-total').innerText = words.length;
            document.getElementById('stat-box1').innerText = words.filter(w => w.box === 1).length;
            document.getElementById('stat-box-mid').innerText = words.filter(w => w.box >= 2 && w.box <= 4).length;
            document.getElementById('stat-box5').innerText = words.filter(w => w.box === 5).length;
        }

        // Load Leitner Card for Review
        function loadNextCard() {
            if(words.length === 0) {
                document.getElementById('no-cards-msg').classList.remove('hidden');
                document.getElementById('card-display').classList.add('hidden');
                document.getElementById('card-actions').classList.add('hidden');
                return;
            }

            // Pick first available word for simplicity
            currentCardIndex = 0;
            const card = words[currentCardIndex];
            currentCardWord = card.german;

            document.getElementById('no-cards-msg').classList.add('hidden');
            document.getElementById('card-display').classList.remove('hidden');
            document.getElementById('card-actions').classList.add('hidden');
            
            document.getElementById('card-box-badge').innerText = 'خانه لایتنر: ' + card.box;
            document.getElementById('card-word').innerText = card.german;
            document.getElementById('card-meaning').innerText = card.persian;
            document.getElementById('card-meaning').classList.add('hidden');
            document.getElementById('card-example').innerText = card.example;
            document.getElementById('card-example').classList.add('hidden');
            document.getElementById('btn-show-answer').classList.remove('hidden');
        }

        function revealAnswer() {
            document.getElementById('card-meaning').classList.remove('hidden');
            document.getElementById('card-example').classList.remove('hidden');
            document.getElementById('btn-show-answer').classList.add('hidden');
            document.getElementById('card-actions').classList.remove('hidden');
        }

        // Leitner Box progression algorithm
        function processAnswer(remembered) {
            if(currentCardIndex === null || !words[currentCardIndex]) return;

            if(remembered) {
                if(words[currentCardIndex].box < 5) words[currentCardIndex].box += 1;
            } else {
                words[currentCardIndex].box = 1;
            }

            words[currentCardIndex].lastReviewed = new Date().toISOString();
            
            // Move reviewed card to the end of the array for cyclic review
            const reviewedCard = words.splice(currentCardIndex, 1)[0];
            words.push(reviewedCard);

            localStorage.setItem('worterbuch_words', JSON.stringify(words));
            updateStats();
            loadNextCard();
        }

        // Native Browser Text-to-Speech (German de-DE)
        function speakGerman(text) {
            if (!('speechSynthesis' in window)) {
                alert('مرورگر شما از قابلیت تلفظ صوتی پشتیبانی نمی‌کند.');
                return;
            }
            window.speechSynthesis.cancel();
            const utterance = new SpeechSynthesisUtterance(text);
            utterance.lang = 'de-DE';
            utterance.rate = 0.9;
            window.speechSynthesis.speak(utterance);
        }

        // Run on load
        window.onload = initApp;
    </script>
